cmake_minimum_required(VERSION 3.19)
project(hypr-edge-trigger VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(BUILD_PLUGIN "Build Hyprland plugin" ON)
if(BUILD_PLUGIN)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(HYPR REQUIRED IMPORTED_TARGET hyprland)
 pkg_check_modules(LUA REQUIRED IMPORTED_TARGET lua)
 add_library(hypr-edge-trigger SHARED src/main.cpp)
 target_link_libraries(hypr-edge-trigger PRIVATE PkgConfig::HYPR PkgConfig::LUA)
 target_compile_options(hypr-edge-trigger PRIVATE -Wall -Wextra -Wno-unused-parameter -fno-gnu-unique)
 target_compile_definitions(hypr-edge-trigger PRIVATE WLR_USE_UNSTABLE)
 install(TARGETS hypr-edge-trigger LIBRARY DESTINATION lib)
endif()
if(BUILD_TESTING)
 add_executable(edge-tests tests/engine.cpp)
 target_include_directories(edge-tests PRIVATE src)
 add_test(NAME edge-state-machine COMMAND edge-tests)
 find_program(LUA_PROGRAM lua)
 if(LUA_PROGRAM)
  add_test(NAME offline-lua-config COMMAND ${LUA_PROGRAM} ${CMAKE_CURRENT_SOURCE_DIR}/tests/local-config.lua ${CMAKE_CURRENT_SOURCE_DIR}/examples/hyprv.lua)
 endif()
endif()
